function normalizeIdentifier(value) { if (value === null || value === undefined) { return ""; } return String(value) .trim() .toUpperCase() .replace(/[\s-]/g, ""); } function objectContainsIdentifier(obj, identifier, type = "vin") { if (!obj || typeof obj !== "object") { return false; } const wanted = normalizeIdentifier(identifier); if (!wanted) { return false; } const keys = type === "lot" ? [ "lot", "lot_number", "lotNumber", "stock", "stock_number", "stockNumber", "stock_number", "StockNumber", "SalvageId", "salvageId", "itemId", "vehicleId" ] : [ "vin", "VIN", "vehicle_vin", "vehicleVin", "slug_vin", "slugVin", "Vin", "vin_number", "vinNumber" ]; for (const key of keys) { if ( Object.prototype.hasOwnProperty.call( obj, key ) ) { const value = normalizeIdentifier(obj[key]); if ( value && value === wanted ) { return true; } } } return false; } function findExactVehicle(data, identifier, type = "vin") { if (!identifier) { return null; } const wanted = normalizeIdentifier(identifier); if (!wanted) { return null; } /* * Przeszukujemy cały wynik API, * również zagnieżdżone obiekty. */ const objects = collectObjects(data); /* * Najpierw szukamy obiektu, * który bezpośrednio zawiera właściwy VIN/LOT. */ for (const obj of objects) { if ( objectContainsIdentifier( obj, identifier, type ) ) { return obj; } } /* * Dodatkowe zabezpieczenie: * jeżeli API ma nietypową strukturę, * sprawdzamy również wszystkie wartości * tekstowe w obiektach. */ for (const obj of objects) { for (const [key, value] of Object.entries(obj)) { if ( typeof value === "string" || typeof value === "number" ) { const normalized = normalizeIdentifier(value); if ( normalized === wanted ) { const lowerKey = String(key).toLowerCase(); if ( type === "vin" && ( lowerKey.includes("vin") ) ) { return obj; } if ( type === "lot" && ( lowerKey.includes("lot") || lowerKey.includes("stock") || lowerKey.includes("salvage") || lowerKey.includes("item") || lowerKey.includes("vehicle") ) ) { return obj; } } } } } return null; } function unwrapResult(json) { if (!json) { return null; } let data = json.data ?? json.result ?? json.vehicle ?? json.car ?? json; /* * Rozpakowanie dodatkowego poziomu data. */ if ( data && data.data && !Array.isArray(data.data) ) { data = data.data; } /* * ========================================================== * NAJWAŻNIEJSZE: * * Jeżeli szukamy VIN-u, MUSIMY znaleźć dokładnie ten VIN. * * Nie wolno już nigdy robić: * * return data[0] * * ponieważ wtedy Mercedes może otworzyć pierwszy samochód. * ========================================================== */ if (VIN) { const exact = findExactVehicle( data, VIN, "vin" ); if (exact) { return exact; } /* * VIN został podany, ale nie znaleźliśmy * dokładnego pojazdu. * * ZWRACAMY NULL. * * Dzięki temu kod nie otworzy przypadkowego * pierwszego samochodu. */ return null; } /* * Analogiczna zasada dla LOT. */ if (LOT) { const exact = findExactVehicle( data, LOT, "lot" ); if (exact) { return exact; } return null; } /* * Jeżeli nie podano ani VIN, ani LOT, * dopiero wtedy możemy zaakceptować pojedynczy * obiekt zwrócony przez API. */ if ( Array.isArray(data) ) { if (data.length === 1) { return data[0]; } return null; } return data; }
